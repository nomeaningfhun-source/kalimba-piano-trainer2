function triggerSound(noteName) { const ctx = getAudioContext(); const freq = FREQUENCIES[noteName]; if (!freq) return; const now = ctx.currentTime; // 1. โน้ตความถี่หลัก (Fundamental Tone) - ตัวเนื้อเสียง const oscMain = ctx.createOscillator(); const gainMain = ctx.createGain(); oscMain.type = 'sine'; oscMain.frequency.setValueAtTime(freq, now); // เลียนแบบการดีด: แตะแรงแล้วค่อยๆ หรี่ลงตามธรรมชาติของลิ้นเหล็ก gainMain.gain.setValueAtTime(0.001, now); gainMain.gain.exponentialRampToValueAtTime(0.7, now + 0.004); // หัวเสียงพุ่งเร็วมาก (Attack) gainMain.gain.exponentialRampToValueAtTime(0.0001, now + 1.8); // หางเสียงกังวานยาว (Decay) // 2. เสียงฮาร์มอนิกแท่งโลหะ (Metallic Overtone) - ทำให้มีประกายแหลมใสแบบเหล็กคาลิมบา const oscOver = ctx.createOscillator(); const gainOver = ctx.createGain(); oscOver.type = 'sine'; // ลิ้นเหล็กคาลิมบามักสั่น overtone ที่ความถี่ ~2.76 เท่าของความถี่หลัก oscOver.frequency.setValueAtTime(freq * 2.76, now); // ฮาร์มอนิกโลหะจะดังแค่ตอนเริ่มดีดแล้วดับไปไวมาก (~0.15 วินาที) gainOver.gain.setValueAtTime(0.25, now); gainOver.gain.exponentialRampToValueAtTime(0.0001, now + 0.15); // 3. เสียงเคาะกระทบเล็บ/เนื้อนิ้ว (Fingertip Click / Thud) const oscClick = ctx.createOscillator(); const gainClick = ctx.createGain(); oscClick.type = 'triangle'; oscClick.frequency.setValueAtTime(freq * 0.5, now); gainClick.gain.setValueAtTime(0.15, now); gainClick.gain.exponentialRampToValueAtTime(0.0001, now + 0.02); // ดับใน 0.02 วิ // เชื่อมต่อระบบเสียงทั้งหมดเข้าลำโพง oscMain.connect(gainMain); gainMain.connect(ctx.destination); oscOver.connect(gainOver); gainOver.connect(ctx.destination); oscClick.connect(gainClick); gainClick.connect(ctx.destination); // สั่งเริ่มและหยุดทำงาน oscMain.start(now); oscOver.start(now); oscClick.start(now); oscMain.stop(now + 1.8); oscOver.stop(now + 0.2); oscClick.stop(now + 0.05); }
